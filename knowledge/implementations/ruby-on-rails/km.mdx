ការអនុវត្ត Patterns ជាក់ស្តែងសម្រាប់ Ruby on Rails 7/8 លើ Ruby 3.3+។ ផ្តោតលើ Faraday Client Timeouts, ការកំណត់ ActiveRecord Connection Pool, Atomic Idempotency Keys, Rails.cache, rack-attack Rules, និង Sidekiq Background Jobs។

## Timeouts first

ក្នុង Rails កំណត់ Database Statement Timeout ក្នុង `config/database.yml` និងកំណត់ Outbound HTTP Timeout លើ HTTP Client៖

```yaml
# config/database.yml
production:
  adapter: postgresql
  encoding: unicode
  pool: <%= ENV.fetch("RAILS_MAX_THREADS") { 5 } %>
  timeout: 5000
  variables:
    statement_timeout: 5000 # ៥ វិនាទីកំណត់ដោយ Postgres
    lock_timeout: 2000      # ២ វិនាទីសម្រាប់ lock wait
```

សម្រាប់ការហៅ Network ទៅខាងក្រៅ ប្រើ Faraday ជាមួយ Timeouts ច្បាស់លាស់៖

```ruby
require 'faraday'

class RatesService
  def self.fetch_rates
    conn = Faraday.new(url: 'https://api.example.com') do |f|
      f.options.timeout = 2.0         # រយៈពេលសរុបអតិបរមា ២ វិនាទី
      f.options.open_timeout = 1.0    # Connect timeout
      f.response :raise_error
      f.adapter Faraday.default_adapter
    end

    response = conn.get('/rates')
    JSON.parse(response.body)
  rescue Faraday::TimeoutError
    Rails.logger.error("Rates service បានផុតម៉ោងកំណត់ ២ វិនាទី")
    raise CustomTimeoutError, "Upstream rates API timed out"
  end
end
```

## Retry with backoff (Faraday Retry)

ប្រើប្រាស់ Middleware Retry របស់ Faraday ជាមួយ Exponential Backoff និង Jitter៖

```ruby
conn = Faraday.new(url: 'https://api.example.com') do |f|
  f.request :retry, {
    max: 3,
    interval: 0.1,
    interval_randomness: 0.5, # Jitter
    backoff_factor: 2,
    exceptions: [Faraday::TimeoutError, Faraday::ConnectionFailed],
    retry_statuses: [408, 429, 500, 502, 503, 504],
    methods: [:get, :head]    # Retry តែ Idempotent Methods ប៉ុណ្ណោះ
  }
  f.adapter Faraday.default_adapter
end
```

## Idempotency keys

ប្រើប្រាស់ PostgreSQL Unique Index និងគ្រប់គ្រងតាមរយៈ `ActiveRecord::Base.transaction`៖

```ruby
class OrdersController < ApplicationController
  def create
    idempotency_key = request.headers['Idempotency-Key']
    return render json: { error: 'តម្រូវឱ្យមាន Idempotency-Key' }, status: :bad_request unless idempotency_key

    ActiveRecord::Base.transaction do
      record = IdempotencyRecord.lock.find_or_initialize_by(key: idempotency_key)

      if record.persisted?
        if record.response_body.present?
          return render json: record.response_body, status: record.status_code
        else
          return render json: { error: 'សំណើកំពុងដំណើរការនៅឡើយ' }, status: :conflict
        end
      end

      record.save!

      order = Order.create!(
        customer_id: params[:customer_id],
        total_cents: params[:total_cents]
      )

      record.update!(
        status_code: 201,
        response_body: order.as_json
      )

      render json: order, status: :created
    end
  rescue ActiveRecord::RecordNotUnique
    render json: { error: 'មានសំណើស្ទួនក្នុងពេលតែមួយ' }, status: :conflict
  end
end
```

## Caching (`Rails.cache`)

ប្រើប្រាស់ `Rails.cache.fetch` ជាមួយ Redis និងកំណត់ `race_condition_ttl` ដើម្បីការពារ Cache Stampede៖

```ruby
class ProductService
  def self.find_product(id)
    Rails.cache.fetch("product:#{id}", expires_in: 5.minutes, race_condition_ttl: 10.seconds) do
      Product.find_by(id: id)
    end
  end
end
```

## Rate limiting (`rack-attack`)

ការពារ Endpoints ពីការ spam ក្នុង `config/initializers/rack_attack.rb`៖

```ruby
class Rack::Attack
  # កម្រិត POST ទៅ /login ត្រឹម ៥ ដងក្នុង ២០ វិនាទីតាម IP
  throttle('logins/ip', limit: 5, period: 20.seconds) do |req|
    req.ip if req.path == '/login' && req.post?
  end

  # កម្រិតទូទៅ ១០០ សំណើក្នុងមួយនាទីសម្រាប់ API
  throttle('api/general', limit: 100, period: 1.minute) do |req|
    if req.path.start_with?('/api')
      req.env['HTTP_AUTHORIZATION'] || req.ip
    end
  end
end
```

## Background jobs ជាមួយ Sidekiq

ដំណើរការ Queue Jobs តែបន្ទាប់ពីទិន្នន័យបាន Save ចូល Database រួចរាល់ដោយប្រើ `after_commit` Hook៖

```ruby
class Order < ApplicationRecord
  after_commit :enqueue_receipt, on: :create

  private

  def enqueue_receipt
    SendReceiptJob.perform_async(id)
  end
end

class SendReceiptJob
  include Sidekiq::Job
  sidekiq_options retry: 5, backoff: :exponential

  def perform(order_id)
    order = Order.find_by(id: order_id)
    return unless order
    return if order.receipt_sent_at.present? # ការពារផ្ញើពីរដង

    # ផ្ញើវិក្កយបត្រ...
    order.update!(receipt_sent_at: Time.current)
  end
end
```
